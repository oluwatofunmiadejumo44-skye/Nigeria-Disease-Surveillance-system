# ---------------- ALGORITHM ----------------
# Percent  = disease cases / all cases in that state * 100
# Growth   = (Cases - PrevCases) / PrevCases * 100
# Alert    = OUTBREAK if Growth >= 50 and Percent >= 10
#            WATCH    if Growth >= 25
#            NORMAL   otherwise
function Get-Analysis($rows) {
    $out = New-Object System.Collections.ArrayList
    foreach ($g in ($rows | Group-Object State)) {
        $total = ($g.Group | Measure-Object -Property Cases -Sum).Sum
        foreach ($r in $g.Group) {
            $cases = [double]$r.Cases; $prev = [double]$r.PrevCases
            $pct = if ($total -gt 0) { [math]::Round($cases / $total * 100, 1) } else { 0 }
            $growth = if ($prev -gt 0) { [math]::Round(($cases - $prev) / $prev * 100, 1) } else { 0 }
            $alert = 'NORMAL'
            if ($growth -ge 50 -and $pct -ge 10) { $alert = 'OUTBREAK' }
            elseif ($growth -ge 25) { $alert = 'WATCH' }
            [void]$out.Add([pscustomobject]@{
                State = $r.State; Zone = $r.Zone; Disease = $r.Disease
                Cases = [int]$cases; Percent = $pct; Growth = $growth; Alert = $alert })
        }
    }
    return $out
}

function Set-Data($csvText) {
    $script:analysis = Get-Analysis ($csvText | ConvertFrom-Csv)
    $cbState.Items.Clear(); [void]$cbState.Items.Add('All states')
    $script:analysis | Select-Object -ExpandProperty State -Unique | Sort-Object |
        ForEach-Object { [void]$cbState.Items.Add($_) }
    $cbState.SelectedIndex = 0
}

function Get-Selection {
    $st = $cbState.SelectedItem; $ds = $cbDisease.SelectedItem
    $script:analysis | Where-Object {
        ($st -eq 'All states' -or $_.State -eq $st) -and ($ds -eq 'All diseases' -or $_.Disease -eq $ds) }
}

function Refresh-View {
    if (-not $script:analysis) { return }
    $sel = @(Get-Selection)
    $dt = New-Object System.Data.DataTable
    foreach ($c in 'State','Zone','Disease','Cases','Percent (%)','Growth (%)','Alert') { [void]$dt.Columns.Add($c) }
    foreach ($r in ($sel | Sort-Object State, @{e='Percent';Descending=$true})) {
        [void]$dt.Rows.Add($r.State, $r.Zone, $r.Disease, $r.Cases, $r.Percent, $r.Growth, $r.Alert)
    }
    $grid.DataSource = $dt
    foreach ($row in $grid.Rows) {
        switch ($row.Cells['Alert'].Value) {
            'OUTBREAK' { $row.DefaultCellStyle.BackColor = [Drawing.Color]::FromArgb(255,200,200) }
            'WATCH'    { $row.DefaultCellStyle.BackColor = [Drawing.Color]::FromArgb(255,240,190) }
        }
    }
    $script:chartData = @($sel | Group-Object Disease | ForEach-Object {
        [pscustomobject]@{ Disease = $_.Name; Cases = ($_.Group | Measure-Object Cases -Sum).Sum } } |
        Sort-Object Cases -Descending)
    $tot = ($script:chartData | Measure-Object Cases -Sum).Sum
    foreach ($c in $script:chartData) { $c | Add-Member Pct ([math]::Round($c.Cases / [math]::Max($tot,1) * 100, 1)) -Force }
    $top = if ($script:chartData.Count) { $script:chartData[0].Disease } else { 'n/a' }
    $out = @($sel | Where-Object Alert -eq 'OUTBREAK').Count
    $lblSummary.Text = "Most prominent: $top     |     Total cases: $tot     |     Outbreak alerts: $out"
    $chart.Invalidate()
}

# Bar chart: share (%) of each disease in the current selection
$chart.Add_Paint({
    param($s, $e)
    $g = $e.Graphics; $g.SmoothingMode = 'AntiAlias'
    $f = New-Object Drawing.Font('Segoe UI', 9)
    $title = New-Object Drawing.Font('Segoe UI', 11, [Drawing.FontStyle]::Bold)
    $g.DrawString('Disease share of reported cases (%)', $title, [Drawing.Brushes]::Black, 10, 10)
    $y = 50; $maxW = $s.Width - 220
    foreach ($d in $script:chartData) {
        $w = [int]($maxW * $d.Pct / 100)
        $g.DrawString($d.Disease, $f, [Drawing.Brushes]::Black, 10, $y + 3)
        $g.FillRectangle([Drawing.Brushes]::SeaGreen, 110, $y, [math]::Max($w,2), 24)
        $g.DrawString("$($d.Pct)%", $f, [Drawing.Brushes]::Black, 115 + $w, $y + 3)
        $y += 38
    }
})
